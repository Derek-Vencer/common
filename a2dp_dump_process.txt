static void inline a2dp_audio_convert_16bit_to_24bit(int32_t *out, int16_t *in, int pcm_len)
{
    for (int i = pcm_len - 1; i >= 0; i--) {
        out[i] = ((int32_t)in[i] << 8);
    }
}
